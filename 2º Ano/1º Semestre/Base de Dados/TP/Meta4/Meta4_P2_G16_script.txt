/* View: VW_AEROPORTO_LOCALIZ                                   */
/*==============================================================*/
create or replace view VW_AEROPORTO_LOCALIZ as
create or replace view VW_AEROPORTO_LOCALIZ as
select
    a.CODAEROPORTO,
    a.NOMEAERPORTO,
    c.NOMECIDADE,
    p.NOMEPAIS
from AEROPORTO a
    left join CIDADE c
        on a.IDCIDADE = c.IDCIDADE
    left join PAIS p
        on a.IDPAIS = p.IDPAIS;

with read only;

/*==============================================================*/
/* View: VW_BAGAGEM_PAX_VOO                                     */
/*==============================================================*/
create or replace view VW_BAGAGEM_PAX_VOO as
create or replace view VW_BAGAGEM_PAX_VOO as
select
    b.RFID,
    b.PESO,
    p.IDCIVIL,
    p.NOME,
    v.CODVOO,
    v.DATAHORAPART
from BAGAGEM b
    left join PASSAGEIRO p
        on b.IDCIVIL = p.IDCIVIL
    left join VOO v
        on b.CODVOO = v.CODVOO;

with read only;

/*==============================================================*/
/* View: VW_INFO_VOO                                            */
/*==============================================================*/
create or replace view VW_INFO_VOO as
select 
    v.CODVOO,
    v.DATAHORAPART,
    v.DATAHORACHEG,
    est.DESCRESTVOO,
    c.NOMEC,
    a.MODELO,
    ao1.NOMEAERPORTO,
    ao2.NOMEAERPORTO
from VOO v
    left join COMPANHIA c on v.SIGALC = c.SIGALC
    left join ESTADOVOO est on v.CODESTVOO = est.CODESTVOO
    left join AERONAVE a on v.IDAERONAVE = a.IDAERONAVE
    left join AEROPORTO ao1 on v.CODAEROPORTO = ao1.CODAEROPORTO
    left join AEROPORTO ao2 on v.AER_CODAEROPORTO = ao2.CODAEROPORTO

with read only;

/*==============================================================*/
/* View: VW_PASSAGEIRO_RESERVA                                  */
/*==============================================================*/
create or replace view VW_PASSAGEIRO_RESERVA as
create or replace view VW_PASSAGEIRO_RESERVA as
select
    p.IDCIVIL,
    p.NOME,
    p.EMAIL,
    p.TELEMOVEL,
    p.PASSAPORTE,
    r.IDRESERVA,
    r.CODVOO,
    r.ESTARESERV
from PASSAGEIRO p
    left join RESERVA r
        on p.IDCIVIL = r.IDCIVIL;

with read only;

/*==============================================================*/
/* View: VW_TRIP_VOO                                            */
/*==============================================================*/
create or replace view VW_TRIP_VOO as
create or replace view VW_TRIP_VOO as
select
    rv.CODVOO,
    t.IDTRIP,
    t.NOMETRIP,
    f.TIPOFUNCAO
from RELATIONSHIP_7 rv
    left join TRIPULANTE t
        on rv.IDTRIP = t.IDTRIP
    left join FUNCAO f
        on t.CODFUNCAO = f.CODFUNCAO;

with read only;
